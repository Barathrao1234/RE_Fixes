            caller_pkg = _file_to_package_early.get(caller_norm, "")
            if caller_pkg:
                same_pkg_fqn = "{}.{}".format(caller_pkg, s)
                for _p in _fqn_to_paths_early.get(same_pkg_fqn, []):
                    if _p not in out:
                        out.append(_p)

            # FIX: If caller has a known package but Class3.java has no package
            # declaration, _fqn_to_paths_early misses the same-package entry.
            # Fall back to physical directory: same-package files live in the
            # same folder. Check if any global candidate for s is in the same
            # directory as caller_file — if yes, treat it as same-package.
            if caller_file:
                _caller_dir = os.path.dirname(os.path.normcase(os.path.abspath(caller_file)))
                for _p in type_to_path_full_early.get(s, []):
                    _p_dir = os.path.dirname(os.path.normcase(os.path.abspath(_p)))
                    if _p_dir == _caller_dir and _p not in out:
                        out.append(_p)

            for _pkg in _file_to_wildcards_early.get(caller_norm, []):
                wfqn = "{}.{}".format(_pkg, s)
                for _p in _fqn_to_paths_early.get(wfqn, []):
                    if _p not in out:
                        out.append(_p)

            for _p in type_to_path_full_early.get(s, []):
                if _p not in out:
                    out.append(_p)

            return out
